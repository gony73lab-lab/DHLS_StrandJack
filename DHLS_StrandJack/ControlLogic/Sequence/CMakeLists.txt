cmake_minimum_required(VERSION 3.16)
project(DHLSIntegratedSequence LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(SequenceTests ../tests/test_sequence_engine.cpp)
add_executable(SimulationServiceTests ../tests/test_simulation.cpp ../EquipmentControlService.cpp)
target_link_libraries(SequenceTests PRIVATE Threads::Threads)
target_link_libraries(SimulationServiceTests PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(SequenceTests PRIVATE /utf-8 /UNDEBUG)
  target_compile_options(SimulationServiceTests PRIVATE /utf-8 /UNDEBUG)
else()
  target_compile_options(SequenceTests PRIVATE -UNDEBUG)
  target_compile_options(SimulationServiceTests PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME SequenceEngine COMMAND SequenceTests)
add_test(NAME SimulationService COMMAND SimulationServiceTests)
